    using System;
    using UnityEngine;
    using UnityEngine.SceneManagement;

    public class GameManager : MonoBehaviour
    {
        [SerializeField] private Health playerHealth;
        [SerializeField] private WaveManager waveManager;
        [SerializeField] private InputReader inputReader;


        public enum GameState {None, Playing, GameOver}
        public GameState state;

        public event Action<GameState> OnStateChange;

        private void Start()
        {
            state = GameState.None;
            OnStateChange?.Invoke(state);

            Cursor.lockState = CursorLockMode.Locked;
        }

        private void OnEnable()
        {
            playerHealth.Dead += OnPlayerDeath;
        }

        private void OnPlayerDeath()
        {
            state = GameState.GameOver;
            waveManager.StopWaves();
            inputReader.enabled = false;
            Cursor.lockState = CursorLockMode.None;
            Cursor.visible = true;
            Time.timeScale = 0f;
            OnStateChange?.Invoke(state);
        }

        public void Restart()
        {
            Time.timeScale = 1f;
            SceneManager.LoadScene(SceneManager.GetActiveScene().buildIndex);
        }
    }

    using UnityEngine;

    public class Projectile : MonoBehaviour
    {
        [SerializeField] private float lifeSpan = 5f;
        private float damage;
        private Rigidbody rb;
        private float expireTime;

        private void Awake()
        {
            rb = GetComponent<Rigidbody>();
            rb.useGravity = false;
        }

        private void Update()
        {
            if(Time.time >  expireTime) Release();

            expireTime -= Time.deltaTime;
        }

        public void Launch(Vector3 dir, float speed, float dmg)
        {
            transform.forward = dir;
            rb.velocity = dir * speed;
            damage = dmg;
            expireTime = Time.time + lifeSpan;
        }

        private void OnTriggerEnter(Collider other)
        {
            if (other.CompareTag("Player")) return;

            if(other.TryGetComponent<Health>(out Health health))
            {
                health.TakeDamage(damage);
            }
            Release();
        }

        private void Release()
        {
            Destroy(gameObject);
        }
    }

    using System;
    using System.Collections;
    using UnityEngine;
    using UnityEngine.AI;

    public class WaveManager : MonoBehaviour
    {
        [SerializeField] private WaveData waveData;
        [SerializeField] private Enemy EnemyPrefab;
        [SerializeField] private Transform player;
        [SerializeField] private EnemyDiedChannel enemyDiedChannel;
        [SerializeField] private Transform[] spawnPoints;
        public event Action<int> EnemyKilled;


        private int currentWave = 0;
        private int enemiesAlive;
        private int enemiesKilled = 0;

        public event Action<int> WaveChanged;
        private Coroutine wavesCoroutine;
        private void OnEnable()
        {
            enemyDiedChannel.Raised += OnEnemyDeath;
        }
        private void OnDisable()
        {
            enemyDiedChannel.Raised -= OnEnemyDeath;
        }

        private void Start()
        {
            wavesCoroutine = StartCoroutine(RunWaves());
        }

        private void OnEnemyDeath(EnemyData enemyData)
        {
            enemiesAlive--;
            enemiesKilled++;
            EnemyKilled?.Invoke(enemiesKilled);
        }

        private IEnumerator RunWaves()
        {
            while (true)
            {
                currentWave++;
                WaveChanged?.Invoke(currentWave);
                int count = waveData.baseEnemyCount + waveData.extraEnemiesPerWave * (currentWave - 1);
                float interval = Mathf.Max(waveData.minSpawnInterval,
                    waveData.baseSpawnInterval - waveData.intervalReductionPerWave * (currentWave - 1));
                for (int i = 0; i < count; i++)
                {
                    SpawnEnemy();
                    yield return new WaitForSeconds(interval);
                }

                yield return new WaitUntil(()=> enemiesAlive == 0);
                yield return new WaitForSeconds(waveData.timeBetweenWaves);
            }
        }

        private void SpawnEnemy()
        {
            SpawnEntry entry = PickAnEntry();
            if (entry == null) return;

            Transform point = spawnPoints[UnityEngine.Random.Range(0, spawnPoints.Length - 1)];

            if (!NavMesh.SamplePosition(point.position, out NavMeshHit hit, 2, NavMesh.AllAreas)) return;

            Enemy enemy = Instantiate(EnemyPrefab, hit.position, Quaternion.identity);
            enemy.Init(entry.enemyData, player);
            enemiesAlive++;
        }

        private SpawnEntry PickAnEntry()
        {
            float total = 0;
            foreach(SpawnEntry entry in waveData.spawnEntries)
            {
                if(currentWave >= entry.unlockWave)
                {
                    total += entry.weight;
                }
            }

            if (total < 0) return null;

            float roll = UnityEngine.Random.Range(0, total);
            SpawnEntry last = null;

            foreach(SpawnEntry entry in waveData.spawnEntries)
            {
                if(entry.unlockWave > currentWave) continue;

                last = entry;
                roll -= entry.weight;
                if (roll < 0) return entry;
            }

            return last;
        }

        public void StopWaves()
        {
            StopCoroutine(wavesCoroutine);
        }
    }

    using UnityEngine;

    public class PlayerShooter : MonoBehaviour
    {
        [SerializeField] private InputReader inputReader;
        [SerializeField] private Projectile projectile;
        [SerializeField] private Transform muzzle;
        [SerializeField] private Transform aim;
        [SerializeField] private float fireCooldown = 0.2f;
        [SerializeField] private float projectileSpeed = 30f;
        [SerializeField] private float damage = 10f;
        [SerializeField] private float maxAimDist = 100f;
        [SerializeField] private LayerMask aimMask;
        private float nextFireTime;

        private void Update()
        {
            if(inputReader.IsFiring && Time.time >= nextFireTime)
            {
                nextFireTime = Time.time + fireCooldown;
                Shoot();
            }
        }

        private void Shoot()
        {
            Vector3 targetPoint;

            if (Physics.Raycast(aim.position, aim.forward, out RaycastHit hit, maxAimDist, aimMask))
            {
                targetPoint = hit.point;
            }
            else
            {
                targetPoint = aim.position + maxAimDist * aim.forward;
            }

            Vector3 direction = (targetPoint - muzzle.position).normalized;

            Projectile projectileObj = Instantiate(projectile, muzzle.position,Quaternion.LookRotation(direction));
            Debug.DrawLine(muzzle.position, targetPoint, Color.green, 1f);
            projectileObj.Launch(direction, projectileSpeed, damage);
        }
    }
